rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function owner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }
    match /users/{userId}/account/deletion {
      allow read: if owner(userId);
      allow create: if owner(userId)
        && request.resource.data.keys().hasOnly(['deleting'])
        && request.resource.data.deleting == true;
      // Immutable marker prevents stale sessions from recreating deleted records.
      allow update, delete: if false;
    }
    match /users/{userId}/{section}/{recordId} {
      allow read, delete: if owner(userId) && section in ['items', 'groups'];
      allow create, update: if owner(userId) && section in ['items', 'groups']
        && !exists(/databases/$(database)/documents/users/$(userId)/account/deletion);
    }
  }
}
